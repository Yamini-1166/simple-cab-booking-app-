// backend.js
const express = require("express");
const mongoose = require("mongoose");
const cors = require("cors");

const app = express();
app.use(cors());
app.use(express.json());

// Connect MongoDB
mongoose.connect("mongodb://localhost:27017/cabapp", { useNewUrlParser: true, useUnifiedTopology: true })
  .then(() => console.log("MongoDB connected"))
  .catch(err => console.log(err));

// Simple Models
const User = mongoose.model("User", new mongoose.Schema({ name: String, email: String, password: String }));
const Ride = mongoose.model("Ride", new mongoose.Schema({ user: String, pickup: String, drop: String, cabType: String, status: String }));

// Routes
app.post("/register", async (req, res) => {
  const user = await User.create(req.body);
  res.json(user);
});

app.post("/login", async (req, res) => {
  const user = await User.findOne({ email: req.body.email, password: req.body.password });
  if (!user) return res.status(400).json({ message: "Invalid credentials" });
  res.json({ message: "Login successful", user });
});

app.post("/book", async (req, res) => {
  const ride = await Ride.create({ ...req.body, status: "booked" });
  res.json(ride);
});

app.get("/rides", async (req, res) => {
  const rides = await Ride.find();
  res.json(rides);
});

const PORT = 5000;
app.listen(PORT, () => console.log(`Backend running on port ${PORT}`));
